cmake_minimum_required(VERSION 3.21)
project(Lua50Runner LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(LUA50_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.sources/lua-5.0.3" CACHE PATH "Lua 5.0.3 source directory")
file(GLOB sources "${LUA50_ROOT}/src/*.c" "${LUA50_ROOT}/src/lib/*.c")
list(FILTER sources EXCLUDE REGEX "ltests.c$")
add_executable(lua50 ${sources} "${LUA50_ROOT}/src/lua/lua.c")
target_include_directories(lua50 PRIVATE "${LUA50_ROOT}/include" "${LUA50_ROOT}/src")
target_compile_definitions(lua50 PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
target_compile_options(lua50 PRIVATE /Od)
