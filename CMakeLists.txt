cmake_minimum_required(VERSION 3.21)
project(AZSSpeedGear VERSION 0.6.2 LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 17)

configure_file(native/toc_notes_identity.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/toc_notes_identity.h" @ONLY)
add_library(AZSSpeedGear SHARED native/backend.cpp native/toc_notes.cpp native/native_update.cpp native/console_bridge.cpp native/diagnostics.cpp native/language_bridge.cpp native/pe_image.cpp native/proxy.cpp "${CMAKE_CURRENT_BINARY_DIR}/proxy.asm" native/proxy.def)
target_include_directories(AZSSpeedGear PRIVATE native/include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(AZSSpeedGear PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(AZSSpeedGear PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_link_libraries(AZSSpeedGear PRIVATE minhook winmm bcrypt)
set_target_properties(AZSSpeedGear PROPERTIES OUTPUT_NAME dwmapi PREFIX "")

target_compile_options(AZSSpeedGearProbe PRIVATE /utf-8 /W4)
