cmake_minimum_required(VERSION 3.21)
project(AZSSpeedGear VERSION 0.2.1 LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
configure_file(native/proxy.asm "${CMAKE_CURRENT_BINARY_DIR}/proxy.asm" COPYONLY)
configure_file(tests/simulation_fixture.asm "${CMAKE_CURRENT_BINARY_DIR}/simulation_fixture.asm" COPYONLY)
add_library(AZSSpeedGear SHARED native/backend.cpp native/language_bridge.cpp native/pe_image.cpp native/proxy.cpp "${CMAKE_CURRENT_BINARY_DIR}/proxy.asm" native/proxy.def)
target_include_directories(AZSSpeedGear PRIVATE native/include)
target_compile_definitions(AZSSpeedGear PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(AZSSpeedGear PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_link_libraries(AZSSpeedGear PRIVATE minhook winmm)
set_target_properties(AZSSpeedGear PROPERTIES OUTPUT_NAME dwmapi PREFIX "")
add_executable(AZSSpeedGearProbe native/probe.cpp native/pe_image.cpp)
target_include_directories(AZSSpeedGearProbe PRIVATE native/include)
target_compile_definitions(AZSSpeedGearProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(AZSSpeedGearProbe PRIVATE /utf-8 /W4)
include(CTest)
if(BUILD_TESTING)
  add_executable(native_core_test tests/native_core_test.cpp)
  target_include_directories(native_core_test PRIVATE native/include)
  target_compile_options(native_core_test PRIVATE /utf-8 /W4)
  add_test(NAME NativeCore COMMAND native_core_test)
  add_executable(language_bridge_test tests/language_bridge_test.cpp native/language_bridge.cpp)
  target_include_directories(language_bridge_test PRIVATE native/include)
  target_compile_definitions(language_bridge_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(language_bridge_test PRIVATE /utf-8 /W4)
  add_test(NAME LanguageBridge COMMAND language_bridge_test)
  find_program(AZSG_POWERSHELL NAMES pwsh powershell REQUIRED)
  add_test(NAME Installer COMMAND "${AZSG_POWERSHELL}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/installer_test.ps1")
  add_executable(clock_host_test tests/clock_host_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/simulation_fixture.asm")
  set_target_properties(clock_host_test PROPERTIES OUTPUT_NAME AZSSpeedGearTestHost)
  target_compile_definitions(clock_host_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(clock_host_test PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
  add_dependencies(clock_host_test AZSSpeedGear)
  add_test(NAME NativeIntegration COMMAND clock_host_test)
  set_tests_properties(NativeIntegration PROPERTIES TIMEOUT 45)
  set(LUA50_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.sources/lua-5.0.3" CACHE PATH "Lua 5.0.3 sources for addon compatibility tests")
  if(EXISTS "${LUA50_ROOT}/src/lapi.c")
    file(GLOB lua_sources "${LUA50_ROOT}/src/*.c" "${LUA50_ROOT}/src/lib/*.c")
    list(FILTER lua_sources EXCLUDE REGEX "ltests.c$")
    add_executable(lua50 ${lua_sources} "${LUA50_ROOT}/src/lua/lua.c")
    target_include_directories(lua50 PRIVATE "${LUA50_ROOT}/include" "${LUA50_ROOT}/src")
    target_compile_definitions(lua50 PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
    # Lua 5.0.3 predates this MSVC optimizer; /O2 miscompiles its table resize loop.
    # This is a test-only interpreter. The native plugin remains optimized.
    target_compile_options(lua50 PRIVATE /Od)
    add_test(NAME Lua50Addon COMMAND lua50 tests/addon_test.lua)
    set_tests_properties(Lua50Addon PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME Lua50Localization COMMAND lua50 tests/localization_test.lua)
    set_tests_properties(Lua50Localization PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
